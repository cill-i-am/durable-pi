import assert from 'node:assert/strict'
import { BACKGROUND_CONTEXT as ctx } from '@earendil-works/chord/context'
import { createModels } from '@earendil-works/pi-ai'
import { Harness,createRegistry } from '@earendil-works/pi-durable'
import { openNodeSqliteStorage } from '@earendil-works/pi-durable/storage/sqlite/node'
import { Effect } from 'effect'
import { fixture } from './support.mts'
import { piPlans as oldPlans } from './legacy-src/planning/adapters/pi.ts'
import { piPlans } from '/Users/cillianbarron/.codex/worktrees/xstate-browser-execution/durable-pi/src/planning/adapters/pi.ts'
const run=Effect.runPromise
for(const cancelled of [false,true]) {
 const f=await fixture()
 const authority={current:()=>Effect.succeed(f.origin),cancelled:()=>Effect.succeed(false),receipts:()=>Effect.succeed([]),evidence:(ref:any)=>Effect.succeed({...ref,state:'valid',detail:'Old HEAD'})}
 const old=oldPlans(async()=>f.harness,authority)
 const receipt=await run(old.update(f.input,f.origin))
 if(cancelled)await run(old.cancel(f.origin.turnId))
 const keys=[['plans.task',f.id],['plans.turn',f.origin.turnId],['plans.operation','tool:1'],['plans.revision',`${f.id}:1`],['plans.revision',`${f.id}:2`]]
 const before=await Promise.all(keys.map(([kind,key])=>f.stored(kind,key)))
 assert.deepEqual(before.filter(Boolean).map(x=>x!.version),cancelled?[1,1,1,1,1]:[1,1,1,1])
 await f.harness.close(ctx)
 const storage=await openNodeSqliteStorage(f.path),harness=await Harness.open(storage,{models:createModels(),registry:createRegistry()},ctx)
 try {
  const current=piPlans(async()=>harness,authority)
  const detail=await run(current.read(f.id))
  assert.deepEqual(detail.plan,JSON.parse(String(before[0]!.value.value)))
  assert.equal(detail.plan.state,cancelled?'cancelled':'completed')
  const replay=await run(current.update(f.input,f.origin).pipe(Effect.result))
  if(cancelled){assert.equal(replay._tag,'Failure');if(replay._tag==='Failure')assert.equal(replay.failure.code,'cancelled')}
  else {assert.equal(replay._tag,'Success');if(replay._tag==='Success')assert.deepEqual(replay.success,receipt)}
  const after=[]
  for(const [kind,key] of keys){const rec=await storage.findDocument({kind,scope:{kind:'session'},key},'current',ctx);after.push(rec?await storage.document(rec.id,'current',ctx):undefined)}
  assert.deepEqual(after.map(x=>x?.record.id),before.map(x=>x?.record.id))
  assert.deepEqual(after[0]!.value.value,before[0]!.value.value)
  assert.deepEqual(after[2]!.value.receipt,before[2]!.value.receipt)
  assert.deepEqual(after.slice(3),before.slice(3))
  console.log(JSON.stringify({probe:'actual-old-HEAD-write-cold-current-read',cancelled,state:detail.plan.state,idsPreserved:true,rawPlanAndReceiptPreserved:true,historyUnchanged:true,versions:after.filter(Boolean).map(x=>x!.version)}))
 }finally{await harness.close(ctx)}
}
