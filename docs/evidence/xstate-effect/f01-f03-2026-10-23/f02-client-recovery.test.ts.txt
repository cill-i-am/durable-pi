import { it as effectIt } from "@effect/vitest"
import { send, waitFor } from "@xstate/effect"
import { Effect, Exit, Scope } from "effect"
import { TestClock } from "effect/testing"
import { describe, expect, it, vi } from "vitest"

import { makeRecovery, importFailure, watchRecovery } from "../src/features/updates/recovery"

const gate = <T,>() => {
  let resolve!: (value: T) => void
  const promise = new Promise<T>((done) => { resolve = done })
  return { promise, resolve }
}
const open = (reload = vi.fn(), version = vi.fn(async (_signal: AbortSignal): Promise<unknown> => null)) => {
  const scope = Effect.runSync(Scope.make())
  const recovery = Effect.runSync(makeRecovery({ currentRevision: "a".repeat(40), reload, version }).pipe(Effect.provideService(Scope.Scope, scope)))
  return { recovery, reload, version, scope, close: () => { recovery.revoke(); return Effect.runPromise(Scope.close(scope, Exit.void)) } }
}
const settled = (recovery: ReturnType<typeof open>["recovery"]) => Effect.runPromise(waitFor(recovery.actor, (state) => state.matches({ version: "idle" })))

describe("explicit client recovery", () => {
  it("retains a different valid build through mixed rollout responses without navigating", async () => {
    const owner = open(), { recovery, reload } = owner
    try {
      for (const revision of [null, {}, "development", "<script>", "a".repeat(40)]) await Effect.runPromise(recovery.observe(revision))
      expect(recovery.snapshot().revision).toBeNull()
      await Effect.runPromise(recovery.observe("b".repeat(40)))
      await vi.waitFor(() => expect(recovery.snapshot().revision).toBe("b".repeat(40)))
      await Effect.runPromise(recovery.observe("a".repeat(40)))
      expect(recovery.snapshot().revision).toBe("b".repeat(40))
      expect(reload).not.toHaveBeenCalled()
    } finally { await owner.close() }
  })
  it("retains independent fences and admits only one explicit reload after all settle", async () => {
    const owner = open(), { recovery, reload } = owner
    try {
      await Effect.runPromise(recovery.block("upload", "Wait for upload"))
      await Effect.runPromise(recovery.block("draft", "Unreadable draft"))
      expect(await Effect.runPromise(recovery.reload())).toBe(false)
      await Effect.runPromise(recovery.block("upload", null))
      await vi.waitFor(() => expect(recovery.snapshot().blocked).toBe("Unreadable draft"))
      expect(await Effect.runPromise(recovery.reload())).toBe(false)
      await Effect.runPromise(recovery.block("draft", null))
      await Effect.runPromise(recovery.connection(false))
      expect(await Effect.runPromise(recovery.reload())).toBe(false)
      await Effect.runPromise(recovery.connection(true))
      expect(await Effect.runPromise(recovery.reload())).toBe(true)
      expect(await Effect.runPromise(recovery.reload())).toBe(false)
      expect(reload).toHaveBeenCalledTimes(1)
    } finally { await owner.close() }
  })
  it("recovers a browser reload rejection and releases subscribers", async () => {
    const owner = open(vi.fn(() => { throw new Error("navigation denied") })), { recovery } = owner
    const listener = vi.fn(), unsubscribe = recovery.subscribe(listener)
    try {
      expect(await Effect.runPromise(recovery.reload())).toBe(false)
      expect(recovery.snapshot().reloading).toBe(false)
      unsubscribe()
      const count = listener.mock.calls.length
      await Effect.runPromise(recovery.connection(false))
      expect(listener).toHaveBeenCalledTimes(count)
    } finally { await owner.close() }
  })
  it.each(["draft", "offline"] as const)("rechecks a late %s fence at actual navigation", async (kind) => {
    const owner = open(), { recovery, reload } = owner
    const unsubscribe = recovery.subscribe(() => {
      if (recovery.snapshot().reloading) Effect.runSync(kind === "draft" ? recovery.block("draft", "Unsaved draft") : recovery.connection(false))
    })
    try {
      expect(await Effect.runPromise(recovery.reload())).toBe(false)
      expect(reload).not.toHaveBeenCalled()
    } finally { unsubscribe(); await owner.close() }
  })
  it("coalesces checks through body completion and ignores ordinary check failures", async () => {
    const response = gate<unknown>(), owner = open(vi.fn(), vi.fn(() => response.promise)), { recovery, version } = owner
    try {
      await Effect.runPromise(recovery.check())
      await vi.waitFor(() => expect(version).toHaveBeenCalledTimes(1))
      for (let i = 0; i < 5; i++) await Effect.runPromise(recovery.check())
      expect(version).toHaveBeenCalledTimes(1)
      response.resolve("b".repeat(40))
      await settled(recovery)
      expect(recovery.snapshot().revision).toBe("b".repeat(40))
      version.mockRejectedValueOnce(new Error("offline service"))
      await Effect.runPromise(recovery.check())
      await vi.waitFor(() => expect(version).toHaveBeenCalledTimes(2))
      await settled(recovery)
      expect(recovery.snapshot().revision).toBe("b".repeat(40))
    } finally { response.resolve(null); await owner.close() }
  })
  it("unmount revokes navigation, removes listeners and joins a late version response without publication", async () => {
    const response = gate<unknown>(), owner = open(vi.fn(), vi.fn(() => response.promise)), { recovery, scope, version } = owner
    const window = new EventTarget() as Window, document = new EventTarget() as Document
    Object.defineProperty(window, "navigator", { value: { onLine: true } })
    Object.defineProperty(document, "visibilityState", { value: "visible" })
    const removeWindow = vi.spyOn(window, "removeEventListener"), removeDocument = vi.spyOn(document, "removeEventListener")
    Effect.runSync(watchRecovery(recovery, window, document).pipe(Effect.forkScoped, Effect.provideService(Scope.Scope, scope)))
    await vi.waitFor(() => expect(version).toHaveBeenCalledTimes(1))
    await Effect.runPromise(recovery.block("draft", "Unsaved"))
    const unloading = new Event("beforeunload", { cancelable: true })
    window.dispatchEvent(unloading)
    expect(unloading.defaultPrevented).toBe(true)
    let closed = false
    const close = owner.close().then(() => { closed = true })
    try {
      await vi.waitFor(() => expect(version.mock.calls[0]![0].aborted).toBe(true))
      expect(closed).toBe(false)
      expect(await Effect.runPromise(recovery.reload())).toBe(false)
      expect(removeWindow).toHaveBeenCalledTimes(5)
      expect(removeDocument).toHaveBeenCalledTimes(1)
      response.resolve("b".repeat(40))
      await close
      expect(recovery.snapshot().revision).toBeNull()
      window.dispatchEvent(new Event("focus"))
      expect(version).toHaveBeenCalledTimes(1)
    } finally { response.resolve(null); await close }
  })
  it("invalid machine facts cannot remove a real reload fence", async () => {
    const owner = open(), { recovery } = owner
    try {
      await Effect.runPromise(recovery.block("draft", "Unreadable"))
      recovery.actor.send({ type: "FACTS", blocked: null, online: "true" } as never)
      expect(await Effect.runPromise(recovery.reload())).toBe(false)
      expect(recovery.snapshot().blocked).toBe("Unreadable")
    } finally { await owner.close() }
  })
  it("recognizes browser asset failures without reclassifying ordinary API or render errors", () => {
    for (const message of ["Failed to fetch dynamically imported module: /assets/settings-old.js", "error loading dynamically imported module", "Importing a module script failed.", "Unable to preload CSS for /assets/old.css", "Loading chunk 18 failed"])
      expect(importFailure(new TypeError(message))).toBe(true)
    for (const error of [new Error("NetworkError"), new TypeError("Failed to fetch"), new Error("Synthetic workspace load failure"), new Error("Cannot read properties of undefined"), { message: "Failed to fetch dynamically imported module" }])
      expect(importFailure(error)).toBe(false)
  })
})

effectIt.effect("check timeout retains occupancy until native cleanup joins and fences its late revision", () => Effect.gen(function* () {
  const response = gate<unknown>()
  const version = vi.fn((_signal: AbortSignal) => response.promise)
  const recovery = yield* makeRecovery({ currentRevision: "a".repeat(40), reload: vi.fn(), version })
  yield* recovery.check()
  yield* waitFor(recovery.actor, (state) => state.matches({ version: { checking: "waiting" } }))
  yield* Effect.promise(async () => { await vi.waitFor(() => expect(version).toHaveBeenCalledTimes(1)) })
  yield* TestClock.adjust(5000)
  yield* waitFor(recovery.actor, (state) => state.matches({ version: { checking: "expired" } }))
  expect(version.mock.calls[0]![0].aborted).toBe(true)
  yield* send(recovery.actor, { type: "CHECK" })
  expect(version).toHaveBeenCalledTimes(1)
  response.resolve("b".repeat(40))
  yield* waitFor(recovery.actor, (state) => state.matches({ version: "idle" }))
  expect(recovery.snapshot().revision).toBeNull()
}).pipe(Effect.scoped))
