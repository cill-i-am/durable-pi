import { Cause, Context, Effect, Exit, Scope } from 'effect'
import { afterEach, expect, it, vi } from 'vitest'
import { makeSkillsOwner, skillsRequests } from '/Users/cillianbarron/.codex/worktrees/xstate-native-ci-budget/durable-pi/src/features/skills/skills-owner'
afterEach(()=>vi.unstubAllGlobals())
it('retains annotated Fail plus native Interrupt after joining a rejected held Skills fetch',async()=>{
 let reject!:(e:unknown)=>void; let signal:AbortSignal|undefined
 const held=new Promise<never>((_,no)=>{reject=no})
 vi.stubGlobal('fetch',vi.fn((_path,init)=>{signal=init.signal;return held}))
 class NativeRequest extends Context.Service<NativeRequest,string>()('F12NativeRequest'){}
 const native=Cause.fromReasons([Cause.makeFailReason('native failed'),Cause.makeInterruptReason(81235).annotate(Context.make(NativeRequest,'f12'))])
 const scope=Effect.runSync(Scope.make())
 const actor=Effect.runSync(makeSkillsOwner('a',skillsRequests('a')).pipe(Effect.provideService(Scope.Scope,scope)))
 await actor.send({type:'COMMAND',id:'one',command:{kind:'select',input:{expectedRevision:1,skills:[]}}})
 await vi.waitFor(()=>expect(signal).toBeDefined())
 let done=false;const closing=Effect.runPromiseExit(Scope.close(scope,Exit.void)).then(v=>{done=true;return v})
 await vi.waitFor(()=>expect(signal?.aborted).toBe(true));expect(done).toBe(false)
 reject(native)
 const result=await closing
 expect(Exit.isFailure(result)).toBe(true)
 if(Exit.isFailure(result)){
  expect(result.cause.reasons.some(r=>r._tag==='Interrupt'&&r.fiberId===81235&&r.annotations.get(NativeRequest.key)==='f12')).toBe(true)
  expect(result.cause.reasons.some(r=>r._tag==='Die'&&r.defect&&typeof r.defect==='object'&&'error' in r.defect&&r.defect.error==='native failed')).toBe(true)
 }
})
